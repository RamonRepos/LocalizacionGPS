classDiagram
    direction LR

    class Usuario {
        +INTEGER id
        +TEXT email
        +TEXT hash_contrasena
        +TEXT fecha_creacion
    }

    class Codigo {
        +TEXT id
        +INTEGER usuario_id
        +TEXT nombre_etiqueta
        +TEXT fecha_creacion
    }

    class Ubicacion {
        +INTEGER id
        +TEXT codigo_id
        +REAL latitud
        +REAL longitud
        +REAL precision_metros
        +TEXT mensaje
        +TEXT fecha_compartida
    }

    class Worker {
        +env DB
        +env ASSETS
        +env SESSION_SECRET
        +main()
        +listen()
    }

    class Aplicacion {
        +crearAplicacion(opciones) Express
        -montarMiddlewareSesion()
        -montarEnrutadores()
        -manejarError(error, peticion, respuesta)
    }

    class BaseDatos {
        -d1 D1Database
        +get(consulta, ...parametros) Promise~Object~
        +all(consulta, ...parametros) Promise~Array~
        +run(consulta, ...parametros) Promise~Resultado~
    }

    class AlmacenAssets {
        +fetch(recurso) Response
        +servirPaginasEstaticas()
    }

    class ServicioSesion {
        -NOMBRE_COOKIE string
        -DURACION_MS number
        +middlewareSesion(secreto) Middleware
        -calcularFirma(valor, secreto) string
        -interpretarValorCookie(valor, secreto) number|null
        -analizarCookies(cabecera) Object
    }

    class ServicioUsuarios {
        +normalizarEmail(email) string
        +esEmailValido(email) boolean
        +hashContrasena(contrasena) string
        +verificarContrasena(contrasena, hashAlmacenado) boolean
    }

    class ServicioQr {
        +generarQrPng(contenido) Promise~Buffer~
        +generarQrPdf(contenido, titulo) Promise~Buffer~
        -agregarEnlace(pagina, documento, url, rectangulo)
        -centrar(texto, fuente, tamano) number
    }

    class EnrutadorAutenticacion {
        +crearEnrutadorAutenticacion(db) Router
        +requerirSesion(peticion, respuesta, siguiente)
    }

    class EnrutadorCodigos {
        +crearEnrutadorCodigos(db) Router
        +postCodigos()
        +getCodigos()
        +getUbicacionesDeCodigo()
        -construirUrlPublica(peticion, codigo) string
        -nuevoIdentificador() string
    }

    class EnrutadorQr {
        +crearEnrutadorQr(db) Router
        +getQrPng()
        +getQrPdf()
        -servirArchivoQr(peticion, respuesta, formato)
    }

    class EnrutadorUbicaciones {
        +crearEnrutadorUbicaciones(db) Router
        +postCompartir()
        +getUbicacionesPropias()
        -superaLimite(clave, ahora) boolean
        -podarRegistrosAntiguos(ahora)
        -coordenadaValida(valor, minimo, maximo) boolean
    }

    class EnrutadorPaginas {
        +crearEnrutadorPaginas(assets) Router
        +getPaginaInicio()
        +getPaginaCompartir(codigo)
        -servirAsset(rutaAsset, respuesta)
    }

    class InterfazPrincipal {
        -iniciar()
        -mostrarAutenticacion()
        -mostrarAplicacion(sesion)
        -alternarPestanas(mostrarLogin)
        -enviarFormularioAutenticacion(url, cuerpo)
        -crearCodigo()
        -cargarCodigos()
        -cargarUbicaciones()
        -cerrarSesion()
        -formatearFecha(fechaIso) string
        -crearElemento(etiqueta, texto) Elemento
    }

    class InterfazCompartir {
        -enviarUbicacion(posicion)
        -falloGps(error)
        -informar(texto)
    }

    Usuario "1" --> "*" Codigo : crea
    Codigo "1" --> "*" Ubicacion : recibe

    Worker --> Aplicacion : crea con db, secreto y assets
    Worker --> BaseDatos : envuelve D1 (binding DB)
    Worker --> AlmacenAssets : binding ASSETS

    Aplicacion --> ServicioSesion : monta middleware
    Aplicacion --> EnrutadorAutenticacion : monta en /api
    Aplicacion --> EnrutadorCodigos : monta en /api
    Aplicacion --> EnrutadorUbicaciones : monta en /api
    Aplicacion --> EnrutadorQr : monta en /qr
    Aplicacion --> EnrutadorPaginas : monta en /

    EnrutadorAutenticacion ..> ServicioUsuarios : usa
    EnrutadorAutenticacion ..> BaseDatos : consulta
    EnrutadorCodigos ..> BaseDatos : consulta
    EnrutadorQr ..> ServicioQr : usa
    EnrutadorQr ..> BaseDatos : consulta
    EnrutadorUbicaciones ..> BaseDatos : consulta
    EnrutadorPaginas ..> AlmacenAssets : sirve HTML

    InterfazPrincipal ..> EnrutadorAutenticacion : consume API
    InterfazPrincipal ..> EnrutadorCodigos : consume API y QR
    InterfazPrincipal ..> EnrutadorUbicaciones : consume API
    InterfazCompartir ..> EnrutadorUbicaciones : consume API
